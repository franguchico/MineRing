cmake_minimum_required(VERSION 3.21)
project(EldenMinecraftBridge LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This bridge requires 64-bit Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
add_library(ermc_shared STATIC src/paths.cpp src/log.cpp src/shm.cpp)
target_include_directories(ermc_shared PUBLIC src include)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
  third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
add_library(dinput8 SHARED src/proxy.cpp src/dinput8.def)
target_link_libraries(dinput8 PRIVATE ermc_shared user32)
set_target_properties(dinput8 PROPERTIES PREFIX "")
add_library(erbridge_core SHARED src/core.cpp src/memutil.cpp src/debugcmd.cpp
  src/frame.cpp src/game.cpp src/compositor.cpp)
target_link_libraries(erbridge_core PRIVATE ermc_shared minhook user32 d3d12 dxgi d3dcompiler)
set_target_properties(erbridge_core PROPERTIES PREFIX "")
add_library(ErmcDepth SHARED src/reshade_addon.cpp)
set_target_properties(ErmcDepth PROPERTIES PREFIX "" SUFFIX ".addon64")
include(CTest)
if(BUILD_TESTING)
  add_executable(ermc_paths_contract tests/paths_contract_test.cpp)
  target_include_directories(ermc_paths_contract PRIVATE src include)
  add_test(NAME ipc_path_contract COMMAND ermc_paths_contract)
  add_executable(ermc_memory_contract tests/memory_contract_test.cpp)
  target_include_directories(ermc_memory_contract PRIVATE src include)
  add_test(NAME native_memory_contract COMMAND ermc_memory_contract)
  add_executable(ermc_movement_safety tests/movement_safety_test.cpp)
  add_test(NAME native_movement_safety COMMAND ermc_movement_safety)
  add_library(ermc_consumer_contract OBJECT tests/consumer_liveness_checks.cpp)
  add_executable(ermc_ipc_probe tests/ipc_probe.cpp)
  target_link_libraries(ermc_ipc_probe PRIVATE ermc_shared)
  add_executable(ermc_guard_probe tests/guard_probe.cpp)
  target_link_libraries(ermc_guard_probe PRIVATE ermc_shared)
  # A synthetic process name exercises the real loader's eligibility checks.
  # It lives only in the test build folder and is never packaged with the mod.
  set_target_properties(ermc_guard_probe PROPERTIES OUTPUT_NAME "eldenring"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/guard-test")
  add_library(ermc_fake_eac SHARED tests/fake_eac.cpp)
  set_target_properties(ermc_fake_eac PROPERTIES OUTPUT_NAME "EasyAntiCheat_EOS"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/guard-test")
  add_executable(ermc_compositor_probe src/compositor_harness.cpp)
  # Production mapping is asynchronous; fixtures must wait for the worker before asserting pixels.
  target_compile_definitions(ermc_compositor_probe PRIVATE ERMC_COMPOSITOR_NONBLOCKING=1)
  target_include_directories(ermc_compositor_probe PRIVATE src include)
  target_link_libraries(ermc_compositor_probe PRIVATE minhook user32 d3d12 dxgi d3dcompiler)
  add_test(NAME native_frame_layout_contract COMMAND ermc_compositor_probe --frame-contract)
endif()
